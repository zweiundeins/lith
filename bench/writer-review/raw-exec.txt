# The benchmark, put in writer/ for the run:

func BenchmarkZZExec(b *testing.B) {
	for _, raw := range []bool{false, true} {
		name := "database-sql"
		if raw {
			name = "raw"
		}
		b.Run(name, func(b *testing.B) {
			db := openDB(b, sqlite.Options{})
			defer db.Close()
			ctx := context.Background()
			conn, err := db.W.Conn(ctx)
			if err != nil {
				b.Fatal(err)
			}
			defer conn.Close()
			const q = "UPDATE counters SET n = n + ? WHERE id = ?"
			st, _ := conn.PrepareContext(ctx, q)
			var ds driver.Stmt
			conn.Raw(func(dc any) error {
				ds, err = dc.(driver.ConnPrepareContext).PrepareContext(ctx, q)
				return err
			})
			args := make([]driver.NamedValue, 2)
			b.ReportAllocs()
			for i := 0; b.Loop(); i++ {
				if i%100 == 0 {
					conn.ExecContext(ctx, "BEGIN")
				}
				if raw {
					err = conn.Raw(func(any) error {
						args[0] = driver.NamedValue{Ordinal: 1, Value: int64(1)}
						args[1] = driver.NamedValue{Ordinal: 2, Value: int64(1 + i%2)}
						_, err := ds.(driver.StmtExecContext).ExecContext(ctx, args)
						return err
					})
				} else {
					_, err = st.ExecContext(ctx, 1, 1+i%2)
				}
				if err != nil {
					b.Fatal(err)
				}
				if i%100 == 99 {
					conn.ExecContext(ctx, "COMMIT")
				}
			}
			conn.ExecContext(ctx, "ROLLBACK")
		})
	}
}
